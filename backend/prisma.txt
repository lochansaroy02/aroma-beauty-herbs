// ============================================================================
//  Barber Syndicate — Prisma schema (reverse-engineered from the Laravel DB)
//  For rebuilding the backend in Next.js.
//
//  Conventions:
//   - Model names are PascalCase; @@map ties each to its real MySQL table.
//   - Column names are kept snake_case to match the existing database exactly.
//   - Money is modelled as Decimal(precision, scale) (the Laravel app uses
//     decimal/double — Decimal is the correct porting choice).
//   - `created_by_id` / `updated_by_id` are audit columns → plain Int? (they
//     reference users but are left un-related to avoid dozens of back-relations).
//   - Laravel framework plumbing (cache, jobs, sessions, password_reset_tokens,
//     personal_access_tokens, failed_jobs) is intentionally omitted — Next.js
//     will use its own auth/queue/session layer.
// ============================================================================

generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider = "mysql"
  url      = env("DATABASE_URL")
}

  is_featured        Boolean     @default(false)
  is_combo           Boolean     @default(false)
  tax_rate           Decimal?    @db.Decimal(8, 2)

// ─────────────────────────── Analytics ───────────────────────────

model PageVisit {
  id            Int       @id @default(autoincrement())
  visitor_id    String?   @db.VarChar(64)   // cookie/hash — for unique visitors
  path          String?   @db.VarChar(255)
  referrer      String?   @db.VarChar(500)
  referrer_host String?   @db.VarChar(191)  // referrer domain, for grouping
  device        String?   @db.VarChar(20)   // mobile | tablet | desktop
  user_agent    String?   @db.VarChar(500)
  ip_hash       String?   @db.VarChar(64)   // salted hash, never the raw IP
  created_at    DateTime? @default(now())

  @@index([visitor_id])
  @@index([created_at])
  @@map("page_visits")
}

// ─────────────────────────── Media (Spatie Media Library) ───────────────────────────
// Polymorphic: (model_type, model_id) points at any owning row. Prisma has no
// native polymorphic relations, so these are plain scalar fields + an index.

model Media {
  id                     Int       @id @default(autoincrement())
  model_type             String
  model_id               Int
  uuid                   String?   @unique
  collection_name        String
  name                   String
  file_name              String
  mime_type              String?
  disk                   String
  conversions_disk       String?
  size                   Int
  manipulations          Json
  custom_properties      Json
  generated_conversions  Json
  responsive_images      Json
  order_column           Int?
  created_at             DateTime?
  updated_at             DateTime?

  @@index([model_type, model_id])
  @@index([order_column])
  @@map("media")
}
